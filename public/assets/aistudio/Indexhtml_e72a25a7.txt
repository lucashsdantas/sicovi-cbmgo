    function handleLoginSubmit(event) {
      event.preventDefault();
      playSound('click');

      const operador = (document.getElementById("login-usuario").value || "").trim();
      const senha = (document.getElementById("login-senha").value || "").trim();
      const lembrar = document.getElementById("login-lembrar").checked;

        exibirModalErroLogin("Por favor, preencha o militar operador e a senha.");
        return;
      }

      if (lembrar) {
        try { localStorage.setItem("ultimoOperador", operador); } catch(e){}
      } else {
        try { localStorage.removeItem("ultimoOperador"); } catch(e){}
      }

      const btn = document.getElementById("btnLoginSubmit");
      btn.innerHTML = `<span class="spinner-sm"></span> Validando Credenciais...`;
      btn.disabled = true;

      // Se estiver rodando sem o backend Apps Script (Preview/Local)
      if (typeof google === 'undefined' || !google.script || !google.script.run) {
        setTimeout(() => {
          playSound('success');
          try {
            sessionStorage.setItem("operadorLogado", operador);
            localStorage.setItem("operadorLogado", operador);
          } catch(e) {}
          btn.innerHTML = `<span class="material-symbols-outlined">login</span><span>ACESSAR SISTEMA</span>`;
          btn.disabled = false;
          exibirDashboard(operador);
        }, 300);
        return;
      }

      // Validação real no Google Apps Script contra a aba OPERADOR da Planilha
      google.script.run
        .withSuccessHandler(function(response) {
          if (response && response.success) {
            playSound('success');
            const usuarioAutenticado = response.usuario || operador;
            try {
              sessionStorage.setItem("operadorLogado", usuarioAutenticado);
              localStorage.setItem("operadorLogado", usuarioAutenticado);

            btn.disabled = false;
            exibirDashboard(usuarioAutenticado);
          } else {
            exibirModalErroLogin((response && response.mensagem) || "Usuário ou senha incorretos!");
          }
        })
        .withFailureHandler(function(err) {
          exibirModalErroLogin("Erro ao comunicar com a planilha: " + (err.message || err));
        })
        .verificarLogin(operador, senha);
    }

    // CARREGAMENTO EM BATCH DOS DADOS DA PLANILHA
    function carregarTudoEmBatch() {
      setSyncStatus("Carregando...", true);

      if (typeof google === 'undefined' || !google.script || !google.script.run) {
        simularDadosMock();
        setSyncStatus("Sincronizado", false);
        return;
      }

      google.script.run
        .withSuccessHandler(function(response) {
          if (!response || !response.success) {
            carregarViaturasFallback();
            carregarTabelaFallback();
            return;
          }

          if (response.viaturasStatus) {
            viaturasStatusMap = response.viaturasStatus;
            exibirViaturas(response.viaturasStatus);
          }
          if (response.viaturasLista) {
            viaturasLista = response.viaturasLista;
            popularDatalist("listaViaturas", response.viaturasLista);
          }
          if (response.militares) {
            listaMilitaresCadastrados = limparListaMilitaresValidos(response.militares);
            popularDatalist("listaCondutores", listaMilitaresCadastrados);
          }
          if (response.bairros) popularDatalist("lista-bairros", response.bairros);
          if (response.destinosPadrao) popularDatalist("listaDestinos", response.destinosPadrao);

          if (response.resumoEscalas) {
            escalasMemoria = response.resumoEscalas;
            renderizarQuadroEscalas(response.resumoEscalas);
          }
          if (response.registros) exibirTabela(response.registros);

          setSyncStatus("Sincronizado", false);
        })
        .withFailureHandler(function(err) {
          console.warn("Batch falhou, usando fallback individual:", err);
          carregarViaturasFallback();
          carregarTabelaFallback();
        })
        .getDadosIniciais();
    }

    function carregarViaturasFallback() {

    }

    function formatarNomeMilitarLimpo(nome) {
      if (!nome || typeof nome !== 'string') return '';
      let str = nome.trim();
      if (!str) return '';
      str = str.replace(/\s*#\d+/g, '').trim();
      str = str.replace(/^(?:CABO|CB)\s+(Cb\b|Cabo\b)/i, '$1');
      str = str.replace(/^(?:SOLDADO|SD)\s+(Sd\b|Soldado\b)/i, '$1');
      str = str.replace(/^(?:SARGENTO|SGT)\s+((?:[123]º\s*)?(?:Sgt\b|Sargento\b))/i, '$1');
      str = str.replace(/^(?:SUBTENENTE|SUB\s*TEN|ST)\s+(St\b|Subten\b)/i, '$1');
      str = str.replace(/^(?:CAPITÃO|CAPITAO|CAP)\s+(Cap\b)/i, '$1');
      str = str.replace(/^(?:MAJOR|MAJ)\s+(Maj\b)/i, '$1');
      str = str.replace(/^(?:TENENTE|TEN)\s+((?:[12]º\s*)?Ten\b)/i, '$1');
      str = str.replace(/^(?:CORONEL|CEL)\s+(Cel\b)/i, '$1');

      if (/^CABO\s+/i.test(str) && !/\bCb\b/i.test(str)) str = str.replace(/^CABO\s+/i, 'Cb ');
      else if (/^SOLDADO\s+/i.test(str) && !/\bSd\b/i.test(str)) str = str.replace(/^SOLDADO\s+/i, 'Sd ');
      else if (/^(?:1º|1)\s*SARGENTO\s+/i.test(str)) str = str.replace(/^(?:1º|1)\s*SARGENTO\s+/i, '1º Sgt ');
      else if (/^(?:2º|2)\s*SARGENTO\s+/i.test(str)) str = str.replace(/^(?:2º|2)\s*SARGENTO\s+/i, '2º Sgt ');
      else if (/^(?:3º|3)\s*SARGENTO\s+/i.test(str)) str = str.replace(/^(?:3º|3)\s*SARGENTO\s+/i, '3º Sgt ');
      else if (/^SARGENTO\s+/i.test(str)) str = str.replace(/^SARGENTO\s+/i, 'Sgt ');
      else if (/^SUBTENENTE\s+/i.test(str)) str = str.replace(/^SUBTENENTE\s+/i, 'Subten ');
      else if (/^CAPIT[ÃA]O\s+/i.test(str)) str = str.replace(/^CAPIT[ÃA]O\s+/i, 'Cap ');
      else if (/^MAJOR\s+/i.test(str)) str = str.replace(/^MAJOR\s+/i, 'Maj ');
      else if (/^(?:1º|1)\s*TENENTE\s+/i.test(str)) str = str.replace(/^(?:1º|1)\s*TENENTE\s+/i, '1º Ten ');
      else if (/^(?:2º|2)\s*TENENTE\s+/i.test(str)) str = str.replace(/^(?:2º|2)\s*TENENTE\s+/i, '2º Ten ');
      else if (/^TENENTE\s+/i.test(str)) str = str.replace(/^TENENTE\s+/i, 'Ten ');
      else if (/^CORONEL\s+/i.test(str)) str = str.replace(/^CORONEL\s+/i, 'Cel ');

      return str.trim();
    }

    function limparListaMilitaresValidos(lista) {
      if (!Array.isArray(lista)) return [];
      const invalidos = ['', 'null', 'undefined', '-', '--', '---', '#n/a', '#ref!', '#valor!', 'nenhum'];
      return lista
        .map(m => (m !== null && m !== undefined ? String(m).trim() : ''))
        .filter(m => m.length > 1 && !invalidos.includes(m.toLowerCase()))
        .map(formatarNomeMilitarLimpo)
        .filter(m => m.length > 1);
    }

    function iniciarAutoRefresh() {
      if (intervalRefresh) clearInterval(intervalRefresh);
      const progressBar = document.getElementById("refresh-progress-bar");
      tempoParaProximoRefresh = 60;
      
      intervalRefresh = setInterval(function() {
        const algumModalAberto = document.querySelector(".modal-backdrop[style*='display: block']");
        
        if (!algumModalAberto) {
          tempoParaProximoRefresh--;
          const porcentagem = ((60 - tempoParaProximoRefresh) / 60) * 100;
          if (progressBar) progressBar.style.width = porcentagem + "%";
          
          if (tempoParaProximoRefresh <= 0) {
            tempoParaProximoRefresh = 60;
            if (progressBar) progressBar.style.width = "0%";
            
            setSyncStatus("Atualizando...", true);
            if (typeof google !== 'undefined' && google.script && google.script.run) {
              google.script.run.withSuccessHandler(function(registros) {
                if (registros) exibirTabela(registros);
                google.script.run.withSuccessHandler(function(status) {
                  if (status) exibirViaturas(status);
                  setSyncStatus("Sincronizado", false);
                }).getViaturasStatus();
              }).carregarDados();
            } else {
              setSyncStatus("Sincronizado", false);
            }
          }
        } else {
          tempoParaProximoRefresh = 60;
          if (progressBar) progressBar.style.width = "0%";
        }
      }, 1000); 
    }

          countIncendio++;

        div.className = `viatura-card ${statusClass}`;
        div.onclick = function () { selecionarViatura(nome, status); };
        div.title = `${nome} - ${statusText} (Clique para ação)`;

          <span class="vtr-name">${nome}</span>
          <span class="vtr-status-label ${statusClass}">${statusText}</span>

    function parseDateTimeFormat(val) {
      if (!val) return null;
      if (val instanceof Date) return isNaN(val.getTime()) ? null : val;
      if (typeof val !== 'string') {
        const d = new Date(val);
        return !isNaN(d.getTime()) ? d : null;
      }
      const s = val.trim();
      if (!s) return null;
      if (s.includes("/")) {
        const parts = s.split(/[\/\s:]/);
        if (parts.length >= 3) {
          const dia = parseInt(parts[0], 10);
          const mes = parseInt(parts[1], 10) - 1;
          const ano = parseInt(parts[2], 10);
          const hora = parts.length > 3 ? parseInt(parts[3], 10) : 0;
          const min = parts.length > 4 ? parseInt(parts[4], 10) : 0;
          const sec = parts.length > 5 ? parseInt(parts[5], 10) : 0;
          const d = new Date(ano, mes, dia, hora, min, sec);
          return !isNaN(d.getTime()) ? d : null;
        }
      }
      const d = new Date(s);
      return !isNaN(d.getTime()) ? d : null;
    }

    function filtrarTabela() {
      const termoGeral = (document.getElementById("filtro-busca-geral").value || "").toLowerCase().trim();
      const relInicio = document.getElementById("filtro-rel-inicio").value;
      const relFim = document.getElementById("filtro-rel-fim").value;
      
      const dtRelInicio = relInicio ? new Date(relInicio) : null;
      const dtRelFim = relFim ? new Date(relFim) : null;

      const colId = (document.getElementById("filtro-col-id")?.value || "").toLowerCase().trim();
      const colVtr = (document.getElementById("filtro-col-viatura")?.value || "").toLowerCase().trim();
      const colCond = (document.getElementById("filtro-col-condutor")?.value || "").toLowerCase().trim();
      const colEqp = (document.getElementById("filtro-col-equipe")?.value || "").toLowerCase().trim();
      const colDest = (document.getElementById("filtro-col-destino")?.value || "").toLowerCase().trim();
      const colBairro = (document.getElementById("filtro-col-bairro")?.value || "").toLowerCase().trim();
      const colRai = (document.getElementById("filtro-col-rai")?.value || "").toLowerCase().trim();
      const colNat = (document.getElementById("filtro-col-natureza")?.value || "").toLowerCase().trim();
      const colOp = (document.getElementById("filtro-col-operador")?.value || "").toLowerCase().trim();

      let somaKm = 0;

      dadosFiltrados = dadosTabela.slice(1).filter(linha => {
        if (!linha) return false;
        
        if (termoGeral) {
          const match = linha.some(val => val && String(val).toLowerCase().includes(termoGeral));
          if (!match) return false;
        }

        if (colId && !String(linha[0] || "").toLowerCase().includes(colId)) return false;
        if (colVtr && !String(linha[1] || "").toLowerCase().includes(colVtr)) return false;
        if (colCond && !String(linha[2] || "").toLowerCase().includes(colCond)) return false;
        if (colEqp && !String(linha[3] || "").toLowerCase().includes(colEqp)) return false;
        if (colDest && !String(linha[4] || "").toLowerCase().includes(colDest)) return false;
        if (colBairro && !String(linha[5] || "").toLowerCase().includes(colBairro)) return false;
        if (colRai && !String(linha[6] || "").toLowerCase().includes(colRai)) return false;
        if (colNat && !String(linha[7] || "").toLowerCase().includes(colNat)) return false;
        if (colOp) {
          const opVal = String(linha[12] || linha[13] || "").toLowerCase();
          if (!opVal.includes(colOp)) return false;
        }

        if (filtroApenasEmAberto) {
          const temChegada = linha[10] && String(linha[10]).trim() !== "";
          if (temChegada) return false;
        }

        if (dtRelInicio || dtRelFim) {
          const dataSaidaStr = linha[8];
          if (!dataSaidaStr) return false;
          const dtSaida = parseDateTimeFormat(dataSaidaStr);
          if (!dtSaida) return false;
          
          if (dtRelInicio && dtSaida < dtRelInicio) return false;
          if (dtRelFim && dtSaida > dtRelFim) return false;
        }

        const kmS = Number(linha[9]);
        const kmC = Number(linha[11]);
        if (!isNaN(kmS) && !isNaN(kmC) && kmC >= kmS) {
          somaKm += (kmC - kmS);
        }

        return true;
      });

      function isLinhaOcorrencia(linha) {
        if (!linha) return false;
        const destino = String(linha[4] || '').toLowerCase();
        const rai = String(linha[6] || '').trim();
        const nat = String(linha[7] || '').toLowerCase().trim();
        if (rai !== '' && rai !== '---') return true;
        if (destino.includes('ocorrência') || destino.includes('ocorrencia') || destino.includes('qrf')) return true;
        if (nat !== '' && nat !== '---' && !nat.includes('abastecimento') && !nat.includes('obm') && !nat.includes('administrativ')) return true;
        return false;
      }

      const totalOcorrencias = dadosFiltrados.filter(isLinhaOcorrencia).length;

      document.getElementById("kpi-total-saidas").textContent = dadosFiltrados.length;
      const elKpiOcorr = document.getElementById("kpi-total-ocorrencias");
      if (elKpiOcorr) elKpiOcorr.textContent = totalOcorrencias;
      document.getElementById("kpi-km-total").textContent = somaKm + " km";

      paginaAtual = 1;
      renderizarTabela();
    }

      if (dadosFiltrados.length === 0) {
        tbody.innerHTML = `
          <tr>
            <td colspan="15" style="text-align: center; padding: 40px; color: var(--slate-400);">
              Nenhum registro encontrado no banco de dados.
            </td>
          </tr>
        `;
        document.getElementById("infoPaginacao").textContent = "Nenhum registro";
        document.getElementById("btnAnterior").disabled = true;
        document.getElementById("btnProximo").disabled = true;
        document.getElementById("label-num-pagina").textContent = "Página 1 de 1";
        return;
      }

      const offset = (paginaAtual - 1) * registrosPorPagina;
      const itens = dadosFiltrados.slice(offset, paginaAtual * registrosPorPagina);
      
      itens.forEach((linha, index) => {

    function aplicarPresetData(tipo) {
      playSound('click');
      document.querySelectorAll(".filter-chip").forEach(c => {
        if (c.id !== "chip-status-em-aberto") c.classList.remove("active");
      });

      const agora = new Date();
      const fusoAgora = new Date(agora.getTime() - agora.getTimezoneOffset() * 60000).toISOString().slice(0, 16);
      
      let inicio = "";
      if (tipo === '24h') {
        const d = new Date(agora.getTime() - 24 * 60 * 60 * 1000);
        inicio = new Date(d.getTime() - d.getTimezoneOffset() * 60000).toISOString().slice(0, 16);
        document.getElementById("chip-24h").classList.add("active");
      } else if (tipo === 'hoje') {
        const d = new Date();
        d.setHours(0, 0, 0, 0);
        inicio = new Date(d.getTime() - d.getTimezoneOffset() * 60000).toISOString().slice(0, 16);
        document.getElementById("chip-hoje").classList.add("active");
      } else if (tipo === '7d') {
        const d = new Date(agora.getTime() - 7 * 24 * 60 * 60 * 1000);
        inicio = new Date(d.getTime() - d.getTimezoneOffset() * 60000).toISOString().slice(0, 16);
        document.getElementById("chip-7d").classList.add("active");
      } else if (tipo === 'tudo') {
        inicio = "";
        document.getElementById("chip-tudo").classList.add("active");
      }

      document.getElementById("filtro-rel-inicio").value = inicio;
      document.getElementById("filtro-rel-fim").value = fusoAgora;
      filtrarTabela();
    }

    function editarRegistro(dados, isManualEdit = false) {
      try {
        limparTodosErros();

        const dataSaidaStr = dados[8]; 
        if (dataSaidaStr) {
          const dtSaida = parseDateTimeFormat(dataSaidaStr);
          if (dtSaida) {
            const diffMs = (new Date()) - dtSaida;
            const diffHrs = diffMs / (1000 * 60 * 60);
            const jaTemChegada = dados[10] && String(dados[10]).trim() !== "";
            if (diffHrs > 48 && (isManualEdit || jaTemChegada)) {
              abrirModalSenhaAdmin(dados, isManualEdit);
              return;
            }
          }
        }

        playSound('click');
        document.getElementById("modal-despacho-title").textContent = `Edição / Retorno do Registro #${dados[0]}`;

        // Em edição / retorno, oculta viaturas em apoio (retorno é estritamente individual)
        viaturasApoioGAS = [];
        const secApoio = document.getElementById("secao-viaturas-apoio-gas");
        if (secApoio) secApoio.style.display = "none";
        renderizarViaturasApoioGAS();

        const campos = ["id", "viatura", "condutor", "equipe", "destino", "bairro", "rai", "natureza", "saida", "kmSaida", "chegada", "kmChegada", "operador"];
        campos.forEach((c, i) => {
          const el = document.getElementById(c);
          if (!el) return;
          if ((c === "saida" || c === "chegada") && dados[i]) {
            const dt = parseDateTimeFormat(dados[i]);
            if (dt) {
              el.value = new Date(dt.getTime() - dt.getTimezoneOffset() * 60000).toISOString().slice(0, 16);
            } else {
              el.value = dados[i];
            }
          } else {
            el.value = dados[i] !== undefined && dados[i] !== null ? dados[i] : "";
          }
        });

        const chegadaExistente = dados[10] ? String(dados[10]).trim() : "";
        if (chegadaExistente === "" && !isManualEdit) {
          definirHoraChegadaAgora();
          setTimeout(() => {
            const kmC = document.getElementById("kmChegada");
            if (kmC) kmC.focus();
          }, 150);
        }

        calcularDistanciaAoDigitar();
        atualizarPreviewEquipe();
        document.getElementById("modalDespacho").style.display = "block";
      } catch(e) {
        console.error("Erro ao abrir modal de edição:", e);
      }
    }

    function preencherCamposPorViatura() {
      const v = document.getElementById("viatura").value.trim().toUpperCase();
      if (!v) {
        atualizarPreviewEquipe();
        return;
      }

      const escalaMatch = escalasMemoria.find(e => String(e.viatura).trim().toUpperCase() === v);
      if (escalaMatch) {
        if (escalaMatch.condutor) document.getElementById("condutor").value = escalaMatch.condutor;
        if (escalaMatch.equipe) document.getElementById("equipe").value = escalaMatch.equipe;
        atualizarPreviewEquipe();
      } else if (typeof google !== 'undefined' && google.script && google.script.run) {
        google.script.run.withSuccessHandler(r => {
          if (r) {
            if (r.condutor) document.getElementById("condutor").value = r.condutor;
            if (r.equipe) document.getElementById("equipe").value = r.equipe;
            atualizarPreviewEquipe();
          }
        }).getEquipePorViatura(v);
      }

      for (let i = 1; i < dadosTabela.length; i++) {
        if (dadosTabela[i] && String(dadosTabela[i][1]).trim().toUpperCase() === v) {
          const ultimoKm = dadosTabela[i][11] || dadosTabela[i][9];
          if (ultimoKm) {
            document.getElementById("kmSaida").value = ultimoKm;
            limparErroCampo(document.getElementById("kmSaida"));
            break;
          }
        }
      }
    }

    function getUltimoKmRegistradoGAS(vtrNome, idAtual) {
      if (!dadosTabela || dadosTabela.length <= 1) return null;
      const vNorm = String(vtrNome || '').trim().toUpperCase();
      let ultimoKm = null;
      let dataMaisRecente = -1;

      dadosTabela.slice(1).forEach(r => {
        if (!r || String(r[0]) === String(idAtual)) return;
        const vLinha = String(r[1] || '').trim().toUpperCase();
        if (vLinha !== vNorm) return;

        const dt = parseDateTimeFormat(r[8]);
        const timestamp = dt ? dt.getTime() : 0;
        const kmC = Number(r[11]);
        const kmS = Number(r[9]);
        const kmVal = (!isNaN(kmC) && kmC > 0) ? kmC : ((!isNaN(kmS) && kmS > 0) ? kmS : null);

        if (kmVal !== null) {
          if (timestamp > dataMaisRecente) {
            dataMaisRecente = timestamp;
            ultimoKm = kmVal;
          } else if (dataMaisRecente <= 0 && (ultimoKm === null || kmVal > ultimoKm)) {
            ultimoKm = kmVal;
          }
        }
      });
      return ultimoKm;
    }

    function getViaturasDisponiveisApoioGAS() {
      const vtrPrincipal = (document.getElementById("viatura").value || "").trim().toUpperCase();
      const termoBusca = (document.getElementById("busca-apoio-gas-input")?.value || "").trim().toUpperCase();
      const todas = viaturasLista && viaturasLista.length > 0 ? viaturasLista : viaturasStatusMap.map(v => v[0]);

      return todas.filter(v => {
        const vNorm = String(v).trim().toUpperCase();
        if (vNorm === vtrPrincipal) return false;
        if (viaturasApoioGAS.some(a => String(a.viatura).trim().toUpperCase() === vNorm)) return false;
        if (termoBusca && !vNorm.includes(termoBusca)) return false;
        return true;
      });
    }

          badge.style.background = "#fef3c7";
          badge.style.color = "#78350f";
          badge.style.borderColor = "#fcd34d";
        } else {
          badge.style.background = "var(--slate-100)";
          badge.style.color = "var(--slate-600)";
          badge.style.borderColor = "var(--slate-200)";
        }
      }

      if (btnConfirmar) {
        btnConfirmar.disabled = (total === 0);
      }

      if (labelConfirmar) {
        labelConfirmar.textContent = total > 0 ? `Adicionar Selecionadas (${total})` : `Adicionar Selecionadas (0)`;

      if (msgFooter) {
        msgFooter.textContent = total > 0 ? `${total} viatura(s) pronta(s) para inclusão` : `Selecione uma ou mais viaturas`;
      }
    }

    function confirmarViaturasApoioGAS() {
      if (viaturasApoioSelecionadasModalGAS.length === 0) {
        fecharModalSeletorApoioGAS();
        return;
      }
      playSound('success');

      viaturasApoioSelecionadasModalGAS.forEach(vNorm => {
        if (viaturasApoioGAS.some(a => String(a.viatura).trim().toUpperCase() === vNorm)) return;

        let cond = "";
        let eqp = "";
        const esc = escalasMemoria.find(e => String(e.viatura).trim().toUpperCase() === vNorm);
        if (esc) {
          cond = esc.condutor || "";
          eqp = esc.equipe || "";
        }

        const ultKm = getUltimoKmRegistradoGAS(vNorm, 'PENDENTE');

        viaturasApoioGAS.push({
          viatura: vNorm,
          condutor: cond,
          equipe: eqp,
          kmSaida: ultKm !== null ? ultKm : ""
        });
      });

      renderizarViaturasApoioGAS();
      fecharModalSeletorApoioGAS();
    }

    // SALVAR REGISTRO
    function salvar() {
      playSound('click');

      const viatura = document.getElementById("viatura");
      const condutor = document.getElementById("condutor");
      const destino = document.getElementById("destino");
      const saida = document.getElementById("saida");
      const kmSaida = document.getElementById("kmSaida");
      const chegada = document.getElementById("chegada");
      const kmChegada = document.getElementById("kmChegada");
      const idAtual = document.getElementById("id").value;
      const isNovoDespacho = (idAtual === "PENDENTE" || !idAtual);

      let hasError = false;

      // Limpar erros prévios
      [viatura, condutor, destino, saida, kmSaida, chegada, kmChegada].forEach(el => {
        if (el) el.classList.remove("input-error");
      });

      if (!viatura.value.trim()) { viatura.classList.add("input-error"); hasError = true; }
      if (!condutor.value.trim()) { condutor.classList.add("input-error"); hasError = true; }
      if (!destino.value.trim()) { destino.classList.add("input-error"); hasError = true; }
      if (!saida.value) { saida.classList.add("input-error"); hasError = true; }
      if (!kmSaida.value || isNaN(Number(kmSaida.value))) { kmSaida.classList.add("input-error"); hasError = true; }

      // Validação estrita de chegada / retorno da viatura principal:
      const temChegada = Boolean(chegada && chegada.value && String(chegada.value).trim() !== "");
      const temKmChegada = Boolean(kmChegada && kmChegada.value !== "" && !isNaN(Number(kmChegada.value)));

      if (temChegada || temKmChegada) {
        if (!temChegada) {
          chegada.classList.add("input-error");
          hasError = true;
          playSound('error');
          alert("Atenção: Para registrar o retorno da viatura, é obrigatório preencher a Data e Hora de Chegada.");
          chegada.focus();
          return;
        }
        if (!temKmChegada) {
          kmChegada.classList.add("input-error");
          hasError = true;
          playSound('error');
          alert("Atenção: Para registrar o retorno da viatura, é obrigatório preencher o KM de Chegada (Odômetro final).");
          kmChegada.focus();
          return;
        }

        const dtS = new Date(saida.value);
        const dtC = new Date(chegada.value);
        if (!isNaN(dtS.getTime()) && !isNaN(dtC.getTime()) && dtC < dtS) {
          chegada.classList.add("input-error");
          playSound('error');
          alert("A data e hora de chegada não pode ser anterior à data e hora de saída.");
          chegada.focus();
          return;
        }
      }

      // Validação de viaturas em apoio
      if (isNovoDespacho && viaturasApoioGAS.length > 0) {
        for (let i = 0; i < viaturasApoioGAS.length; i++) {
          const vA = viaturasApoioGAS[i];
          if (!vA.viatura) {
            alert(`Viatura em apoio #${i + 1} inválida.`);
            return;
          }
          if (!vA.condutor || !vA.condutor.trim()) {
            alert(`Por favor, preencha o condutor da viatura em apoio ${vA.viatura}.`);
            return;
          }
          if (vA.kmSaida === "" || isNaN(Number(vA.kmSaida))) {
            alert(`Por favor, preencha o KM de saída da viatura em apoio ${vA.viatura}.`);
            return;
          }
        }
      }

      if (hasError) {
        playSound('error');
        alert("Preencha todos os campos obrigatórios destacados em vermelho.");
        return;
      }

      const r = {
        ID: idAtual,
        viatura: viatura.value.trim().toUpperCase(),
        condutor: condutor.value.trim(),
        equipe: document.getElementById("equipe").value.trim(),
        destino: destino.value.trim(),
        bairro: document.getElementById("bairro").value.trim(),
        rai: document.getElementById("rai").value.trim(),
        natureza: document.getElementById("natureza").value.trim(),
        dataHoraSaida: saida.value,
        kmSaida: kmSaida.value,
        dataHoraChegada: document.getElementById("chegada").value,
        kmChegada: document.getElementById("kmChegada").value,
        operador: document.getElementById("operador").value || getOperadorAtual() || "Operador"
      };

      let listaPayloads = [r];
      if (isNovoDespacho && viaturasApoioGAS.length > 0) {
        const apoiosPayload = viaturasApoioGAS.map(vA => ({
          ID: "PENDENTE",
          viatura: vA.viatura.trim().toUpperCase(),
          condutor: vA.condutor.trim(),
          equipe: (vA.equipe || "").trim(),
          destino: r.destino,
          bairro: r.bairro,
          rai: r.rai,
          natureza: r.natureza,
          dataHoraSaida: r.dataHoraSaida,
          kmSaida: vA.kmSaida,
          dataHoraChegada: "", // Retorno registrado individualmente por viatura
          kmChegada: "",
          operador: r.operador
        }));
        listaPayloads = [r, ...apoiosPayload];
      }

      // Verificação de divergência de KM
      const avisos = [];
      const ultimoKm = getUltimoKmRegistradoGAS(r.viatura, r.ID);
      const kmSNum = Number(r.kmSaida);
      const kmCNum = r.kmChegada !== "" && !isNaN(Number(r.kmChegada)) ? Number(r.kmChegada) : null;

      if (ultimoKm !== null && !isNaN(kmSNum)) {
        if (kmSNum < ultimoKm) {
          avisos.push(`Viatura ${r.viatura}: O KM de saída informado (<strong>${kmSNum.toLocaleString('pt-BR')} km</strong>) é <strong>MENOR</strong> que o último KM registrado (${ultimoKm.toLocaleString('pt-BR')} km).`);
        } else if ((kmSNum - ultimoKm) > 100) {
          avisos.push(`Viatura ${r.viatura}: O KM de saída informado (<strong>${kmSNum.toLocaleString('pt-BR')} km</strong>) é mais de 100 km superior ao último KM registrado (${ultimoKm.toLocaleString('pt-BR')} km — diferença de <strong>${(kmSNum - ultimoKm).toLocaleString('pt-BR')} km</strong>).`);
        }
      }

      if (kmCNum !== null && !isNaN(kmSNum)) {
        if (kmCNum < kmSNum) {
          avisos.push(`Viatura ${r.viatura}: O KM de chegada informado (<strong>${kmCNum.toLocaleString('pt-BR')} km</strong>) é <strong>MENOR</strong> que o KM de saída (${kmSNum.toLocaleString('pt-BR')} km).`);
        } else if ((kmCNum - kmSNum) > 100) {
          avisos.push(`Viatura ${r.viatura}: A distância percorrida informada nesta saída (<strong>${(kmCNum - kmSNum).toLocaleString('pt-BR')} km</strong>) é <strong>SUPERIOR a 100 km</strong>.`);
        }
      }

      // Verificação de KM para as viaturas em apoio
      if (isNovoDespacho && viaturasApoioGAS.length > 0) {
        viaturasApoioGAS.forEach(vA => {
          const ultKmApoio = getUltimoKmRegistradoGAS(vA.viatura, 'PENDENTE');
          const kmSNumApoio = Number(vA.kmSaida);
          if (ultKmApoio !== null && !isNaN(kmSNumApoio)) {
            if (kmSNumApoio < ultKmApoio) {
              avisos.push(`Viatura em apoio ${vA.viatura}: O KM de saída informado (<strong>${kmSNumApoio.toLocaleString('pt-BR')} km</strong>) é <strong>MENOR</strong> que o último KM registrado (${ultKmApoio.toLocaleString('pt-BR')} km).`);
            } else if (kmSNumApoio - ultKmApoio > 100) {
              avisos.push(`Viatura em apoio ${vA.viatura}: O KM de saída informado (<strong>${kmSNumApoio.toLocaleString('pt-BR')} km</strong>) é mais de 100 km superior ao último KM registrado (${ultKmApoio.toLocaleString('pt-BR')} km).`);
            }
          }
        });
      }

      if (avisos.length > 0) {
        dadosPendentesGravacao = listaPayloads.length > 1 ? listaPayloads : r;
        const containerAvisos = document.getElementById("conteudo-avisos-km");
        if (containerAvisos) {
          containerAvisos.innerHTML = `<ul style="margin: 0; padding-left: 20px;">` + avisos.map(a => `<li style="margin-bottom: 6px;">${a}</li>`).join("") + `</ul>`;
        }
        document.getElementById("modalConfirmacaoKm").style.display = "block";
        return;
      }

      if (listaPayloads.length > 1) {
        executarGravacaoMultiplaGAS(listaPayloads);
      } else {
        executarGravacaoGAS(r);
      }
    }

    function executarGravacaoGAS(r) {
      const btnSalvar = document.getElementById("btnSalvar");
      const textoOriginal = btnSalvar.innerHTML;
      btnSalvar.innerHTML = `<span class="spinner-sm"></span> Gravando...`;
      btnSalvar.disabled = true;

      // Fecha modal imediatamente para liberar o operador e direcionar a atenção para a tabela
      fecharModal();
      btnSalvar.innerHTML = textoOriginal;
      btnSalvar.disabled = false;

      // Ativa barra de progresso intuitiva no topo da tabela
      iniciarProgressoTabela(`Gravando dados da viatura ${r.viatura} no banco de dados...`, 35);

      if (typeof google === 'undefined' || !google.script || !google.script.run) {
        let idGeradoTemp = r.ID === "PENDENTE" ? (dadosTabela.length > 1 ? Number(dadosTabela[1][0]) + 1 : 1) : r.ID;
        const rowFormatada = [
          idGeradoTemp, r.viatura, r.condutor, r.equipe, r.destino, r.bairro, r.rai, r.natureza,
          formatarDataExibicao(r.dataHoraSaida), r.kmSaida, formatarDataExibicao(r.dataHoraChegada), r.kmChegada, r.operador
        ];
        if (r.ID === "PENDENTE") {
          if (dadosTabela.length > 0) dadosTabela.splice(1, 0, rowFormatada);
        } else {
          const idx = dadosTabela.findIndex(row => row && String(row[0]) === String(r.ID));
          if (idx !== -1) dadosTabela[idx] = rowFormatada;
        }
        try {
          localStorage.setItem("sicovi_local_bd_registros", JSON.stringify(dadosTabela));
        } catch(e) {}
        const novoStatusLocal = (!r.dataHoraChegada || String(r.dataHoraChegada).trim() === "") ? "EMPENHADA" : "DISPONÍVEL";
        const vtrObj = viaturasStatusMap.find(v => v && String(v[0]).trim().toUpperCase() === r.viatura);
        if (vtrObj) vtrObj[2] = novoStatusLocal;
        exibirViaturas(viaturasStatusMap);

        ultimoIdGravado = idGeradoTemp;
        filtrarTabela();

        concluirProgressoTabela(`✓ Registro #${idGeradoTemp} gravado e tabela atualizada!`, true);
        exibirToastSucesso("Registro Salvo!", `Registro ID ${idGeradoTemp} armazenado com sucesso no banco de dados.`);
        return;
      }

      // Modo Google Apps Script real conectado à Planilha Google Sheets
      google.script.run
        .withSuccessHandler(function(response) {
          playSound('success');
          const idFinal = (response && response.id) ? response.id : (typeof response === 'number' ? response : r.ID);
          ultimoIdGravado = idFinal;

          avancarProgressoTabela(`Registro #${idFinal} gravado! Atualizando tabela em tempo real...`, 75);

          // 1. Atualização OTIMISTA IMEDIATA na tabela (0ms de espera visual)
          let linhaParaTabela = null;
          if (response && response.registro && Array.isArray(response.registro)) {
            linhaParaTabela = response.registro;
          } else {
            linhaParaTabela = [
              idFinal, r.viatura, r.condutor, r.equipe, r.destino, r.bairro, r.rai, r.natureza,
              formatarDataExibicao(r.dataHoraSaida), r.kmSaida, formatarDataExibicao(r.dataHoraChegada), r.kmChegada, r.operador
            ];
          }

          if (r.ID === "PENDENTE" || !r.ID) {
            if (dadosTabela.length > 0) dadosTabela.splice(1, 0, linhaParaTabela);
          } else {
            const idx = dadosTabela.findIndex(row => row && String(row[0]) === String(idFinal));
            if (idx !== -1) dadosTabela[idx] = linhaParaTabela;
            else if (dadosTabela.length > 0) dadosTabela.splice(1, 0, linhaParaTabela);
          }

          // 2. Atualização imediata do status da viatura
          const novoStatus = (response && response.novoStatusVtr) ? response.novoStatusVtr : ((!r.dataHoraChegada || String(r.dataHoraChegada).trim() === "") ? "EMPENHADA" : "DISPONÍVEL");
          const vtrObj = viaturasStatusMap.find(v => v && String(v[0]).trim().toUpperCase() === r.viatura);
          if (vtrObj) vtrObj[2] = novoStatus;
          exibirViaturas(viaturasStatusMap);

          // 3. Renderiza a tabela instantaneamente na tela!
          filtrarTabela();

          // 4. Confirmação e sincronização em segundo plano com a planilha
          google.script.run
            .withSuccessHandler(function(registrosAtualizados) {
              if (registrosAtualizados && registrosAtualizados.length > 0) {
                exibirTabela(registrosAtualizados);
              }
              concluirProgressoTabela(`✓ Registro #${idFinal} gravado e tabela 100% atualizada no banco de dados!`, true);
              google.script.run.withSuccessHandler(exibirViaturas).getViaturasStatus();
            })
            .withFailureHandler(function() {
              concluirProgressoTabela(`✓ Registro #${idFinal} gravado com sucesso no banco de dados!`, true);
            })
            .carregarDados();

          exibirToastSucesso("Registro Salvo!", `Ocorrência ID ${idFinal} gravada e sincronizada com sucesso no banco de dados.`);
        })
        .withFailureHandler(function(err) {
          playSound('error');
          falhaProgressoTabela("Erro ao gravar no banco de dados: " + (err.message || err));
          alert("Erro ao gravar no banco de dados: " + (err.message || err));
        })
        .salvarRegistro(r);
    }

    function executarGravacaoMultiplaGAS(listaDespachos) {
      const btnSalvar = document.getElementById("btnSalvar");
      const textoOriginal = btnSalvar.innerHTML;
      btnSalvar.innerHTML = `<span class="spinner-sm"></span> Gravando Comboio...`;
      btnSalvar.disabled = true;

      // Fecha modal imediatamente para liberar o operador e direcionar a atenção para a tabela
      fecharModal();
      btnSalvar.innerHTML = textoOriginal;
      btnSalvar.disabled = false;

      // Ativa barra de progresso intuitiva no topo da tabela
      iniciarProgressoTabela(`Gravando acionamento simultâneo de ${listaDespachos.length} viaturas no banco de dados...`, 30);

      if (typeof google === 'undefined' || !google.script || !google.script.run) {
        let ultimoIdNum = 0;
        for (let i = 1; i < dadosTabela.length; i++) {
          const rowId = Number(dadosTabela[i][0]);
          if (!isNaN(rowId) && rowId > ultimoIdNum) ultimoIdNum = rowId;
        }

        const idsGerados = [];
        listaDespachos.forEach((d, idx) => {
          const novoId = ultimoIdNum + idx + 1;
          idsGerados.push(novoId);
          const rowFormatada = [
            novoId, d.viatura, d.condutor, d.equipe, d.destino, d.bairro, d.rai, d.natureza,
            formatarDataExibicao(d.dataHoraSaida), d.kmSaida, formatarDataExibicao(d.dataHoraChegada), d.kmChegada, d.operador
          ];
          if (dadosTabela.length > 0) dadosTabela.splice(1, 0, rowFormatada);

          const vtrObj = viaturasStatusMap.find(v => v && String(v[0]).trim().toUpperCase() === d.viatura);
          if (vtrObj) vtrObj[2] = "EMPENHADA";
        });

        try {
          localStorage.setItem("sicovi_local_bd_registros", JSON.stringify(dadosTabela));
        } catch(e) {}

        exibirViaturas(viaturasStatusMap);
        ultimoIdGravado = idsGerados[idsGerados.length - 1];
        filtrarTabela();

        concluirProgressoTabela(`✓ ${listaDespachos.length} viaturas acionadas com sucesso (IDs: ${idsGerados.join(', ')})!`, true);
        exibirToastSucesso("Acionamento Simultâneo Salvo!", `${listaDespachos.length} saídas registradas com sucesso no banco de dados.`);
        return;
      }

      // Modo Google Apps Script real conectado à Planilha Google Sheets
      google.script.run
        .withSuccessHandler(function(response) {
          playSound('success');
          const idsSalvos = (response && response.ids) ? response.ids : [];
          const registrosSalvos = (response && response.registros) ? response.registros : [];

          avancarProgressoTabela(`Acionamento de ${listaDespachos.length} viaturas gravado! Atualizando tabela...`, 75);

          // Atualização Otimista imediata na tabela
          if (registrosSalvos.length > 0) {
            registrosSalvos.forEach(reg => {
              if (dadosTabela.length > 0) dadosTabela.splice(1, 0, reg);
            });
          } else {
            listaDespachos.forEach((d, idx) => {
              const idFinal = idsSalvos[idx] || (typeof d.ID === 'number' ? d.ID : "OK");
              const linhaParaTabela = [
                idFinal, d.viatura, d.condutor, d.equipe, d.destino, d.bairro, d.rai, d.natureza,
                formatarDataExibicao(d.dataHoraSaida), d.kmSaida, formatarDataExibicao(d.dataHoraChegada), d.kmChegada, d.operador
              ];
              if (dadosTabela.length > 0) dadosTabela.splice(1, 0, linhaParaTabela);
            });
          }

          // Atualização imediata do status das viaturas para EMPENHADA
          listaDespachos.forEach(d => {
            const vtrObj = viaturasStatusMap.find(v => v && String(v[0]).trim().toUpperCase() === d.viatura);
            if (vtrObj) vtrObj[2] = "EMPENHADA";
          });
          exibirViaturas(viaturasStatusMap);

          if (idsSalvos.length > 0) ultimoIdGravado = idsSalvos[idsSalvos.length - 1];
          filtrarTabela();

          // Confirmação e sincronização em segundo plano com a planilha
          google.script.run
            .withSuccessHandler(function(registrosAtualizados) {
              if (registrosAtualizados && registrosAtualizados.length > 0) {
                exibirTabela(registrosAtualizados);
              }
              concluirProgressoTabela(`✓ Acionamento de ${listaDespachos.length} viaturas gravado e tabela 100% atualizada!`, true);
              google.script.run.withSuccessHandler(exibirViaturas).getViaturasStatus();
            })
            .withFailureHandler(function() {
              concluirProgressoTabela(`✓ ${listaDespachos.length} saídas gravadas com sucesso no banco de dados!`, true);
            })
            .carregarDados();

          exibirToastSucesso("Acionamento Simultâneo Registrado!", `${listaDespachos.length} viaturas despachadas (IDs: ${idsSalvos.join(', ')}). Chegadas serão registradas individualmente.`);
        })
        .withFailureHandler(function(err) {
          playSound('error');
          falhaProgressoTabela("Erro ao gravar acionamento simultâneo: " + (err.message || err));
          alert("Erro ao gravar acionamento simultâneo no banco de dados: " + (err.message || err));
        })
        .salvarMultiplosDespachos(listaDespachos);
    }

    function formatarDataExibicao(isoStr) {
      if (!isoStr) return "";
      try {
        const d = new Date(isoStr);
        if (isNaN(d.getTime())) return isoStr;
        const pad = (n) => n < 10 ? '0' + n : n;
        return `${pad(d.getDate())}/${pad(d.getMonth()+1)}/${d.getFullYear()} ${pad(d.getHours())}:${pad(d.getMinutes())}`;
      } catch (e) {
        return isoStr;
      }
    }

    function renderizarStripViaturasEscala() {
      const stripEl = document.getElementById("escala-viaturas-pill-strip");
      if (!stripEl) return;
      stripEl.innerHTML = "";

      const todasVtrs = viaturasLista && viaturasLista.length > 0 ? viaturasLista : viaturasStatusMap.map(v => v[0]);
      if (!todasVtrs || todasVtrs.length === 0) return;

      const gruposCategorias = [
        { id: 'RESGATE', label: 'Resgate (UR / APH)', icon: 'airport_shuttle' },
        { id: 'INCÊNDIO', label: 'Incêndio & Salvamento (ABT / ABS)', icon: 'fire_truck' },
        { id: 'ADM', label: 'Apoio & Administrativas (ASA / ADM)', icon: 'minor_crash' }
      ];

      gruposCategorias.forEach(catObj => {
        if (filtroCategoriaEscala !== 'todas') {
          if (filtroCategoriaEscala === 'RESGATE' && catObj.id !== 'RESGATE') return;
          if (filtroCategoriaEscala.includes('INC') && catObj.id !== 'INCÊNDIO') return;
          if (filtroCategoriaEscala === 'ADM' && catObj.id !== 'ADM') return;
        }

        const vtrsDestaCat = todasVtrs.filter(vtr => {
          const c = getCategoriaViatura(vtr);
          if (catObj.id === 'RESGATE') return c === "RESGATE" || vtr.startsWith("UR");
          if (catObj.id === 'INCÊNDIO') return c === "INCÊNDIO" || c === "INCENDIO" || c === "SALVAMENTO" || c === "INCÊNDIO/SALVAMENTO" || vtr.startsWith("ABT") || vtr.startsWith("ABS");
          return c !== "RESGATE" && !vtr.startsWith("UR") && c !== "INCÊNDIO" && c !== "INCENDIO" && c !== "SALVAMENTO" && c !== "INCÊNDIO/SALVAMENTO" && !vtr.startsWith("ABT") && !vtr.startsWith("ABS");
        });

        if (vtrsDestaCat.length === 0) return;

        const catWrap = document.createElement("div");
        catWrap.className = "escala-cat-group-wrapper";
        catWrap.innerHTML = `
          <div class="escala-cat-group-header">
            <span class="material-symbols-outlined" style="font-size: 16px;">${catObj.icon}</span>
            <span>${catObj.label}</span>
            <span class="escala-cat-counter">${vtrsDestaCat.length}</span>
          </div>
          <div class="escala-cat-pills-row"></div>
        `;

        const pillsRow = catWrap.querySelector(".escala-cat-pills-row");

        vtrsDestaCat.forEach(vtr => {
          const match = escalasMemoria.find(e => String(e.viatura).trim().toUpperCase() === String(vtr).trim().toUpperCase());
          const temCondutor = match && match.condutor && match.condutor.trim() !== "";
          const qtdEquipe = match && match.equipe ? match.equipe.split(",").filter(s => s.trim().length > 0).length : 0;
          
          let badgeClass = "vazia";
          let badgeLabel = "Sem Escala";
          if (temCondutor && qtdEquipe > 0) {
            badgeClass = "completa";
            badgeLabel = `Completa (${1 + qtdEquipe} mil.)`;
          } else if (temCondutor || qtdEquipe > 0) {
            badgeClass = "parcial";
            badgeLabel = temCondutor ? "Só Condutor" : `${qtdEquipe} Equipe`;
          }

          const isAtiva = escalaViaturaAtiva === vtr;

          const pill = document.createElement("div");
          pill.className = `escala-vtr-pill ${isAtiva ? 'active' : ''}`;
          pill.onclick = function() { selecionarViaturaParaEscala(vtr); };

          pill.innerHTML = `
            <div style="display:flex; flex-direction:column; text-align:left;">
              <strong style="font-size:0.85rem; color:${isAtiva ? '#ffffff' : 'var(--slate-900)'};">${vtr}</strong>
              <span class="vtr-pill-badge ${badgeClass}" style="${isAtiva ? 'background:rgba(255,255,255,0.25); color:#ffffff; border-color:transparent;' : ''}">${badgeLabel}</span>
            </div>
          `;
          pillsRow.appendChild(pill);
        });

        stripEl.appendChild(catWrap);
      });
    }

    function renderizarMilitaresDisponiveisEscala() {

      const busca = (document.getElementById("escala-militar-busca")?.value || "").toLowerCase().trim();

      if (filtrados.length === 0) {
        container.innerHTML = `<div style="padding: 24px; text-align: center; color: var(--slate-400);">Nenhum militar encontrado.</div>`;
        return;
      }

      filtrados.forEach(militarNome => {
        const ehCondutorAtivo = escalaCondutorAtivo === militarNome;
        const ehCombatenteAtivo = escalaCombatentesAtivos.includes(militarNome);

        let atribuicaoGeral = "";

            atribuicaoGeral = `${esc.viatura} (Condutor)`;

          }
          if (esc.equipe && esc.equipe.toLowerCase().includes(militarNome.toLowerCase())) {
            atribuicaoGeral = `${esc.viatura} (Equipe)`;
            break;
          }

        card.innerHTML = `

            <span style="font-size:0.86rem; font-weight:700; color:var(--slate-800);">${militarNome}</span>

    function salvarEscala(irParaProxima = false) {
      playSound('click');
      const vtr = escalaViaturaAtiva;
      if (!vtr) return;

      const btn = document.getElementById("btnSalvarEscala");
      btn.disabled = true;
      btn.innerHTML = `<span class="spinner-sm"></span> Gravando no Banco de Dados...`;

      const d = {
        viatura: vtr,
        condutor: escalaCondutorAtivo.trim(),
        equipe: escalaCombatentesAtivos.join(", ").trim()
      };

      const idx = escalasMemoria.findIndex(e => String(e.viatura).trim().toUpperCase() === vtr.toUpperCase());

      else escalasMemoria.push(d);
      
      renderizarQuadroEscalas(escalasMemoria);
      renderizarStripViaturasEscala();

    // Formatar data em cima e hora embaixo (duas linhas) no relatório PDF
    function formatarDataHoraDuasLinhas(val) {
      if (!val || String(val).trim() === '') return '';
      const s = String(val).trim();
      if (s.toUpperCase() === 'EM ABERTO') return 'EM ABERTO';

      if (s.includes(' ')) {
        const parts = s.split(' ');
        if (parts.length >= 2 && parts[0].includes('/') && parts[1].includes(':')) {
          return `${parts[0]}\n${parts[1]}`;
        }
      }
      if (s.includes('T')) {
        const d = new Date(s);
        if (!isNaN(d.getTime())) {
          const pad = n => (n < 10 ? '0' + n : String(n));
          return `${pad(d.getDate())}/${pad(d.getMonth() + 1)}/${d.getFullYear()}\n${pad(d.getHours())}:${pad(d.getMinutes())}`;
        }
      }
      return s;
    }

    // Formatar equipe em duas linhas para economizar largura horizontal
    function formatarEquipeDuasLinhas(val) {
      if (!val || String(val).trim() === '') return '';
      const partes = String(val).split(',').map(s => s.trim()).filter(Boolean);
      if (partes.length <= 1) return partes[0] || '';
      const meio = Math.ceil(partes.length / 2);
      const l1 = partes.slice(0, meio).join(', ');
      const l2 = partes.slice(meio).join(', ');
      return `${l1}\n${l2}`;
    }

    // Formatar natureza em duas linhas para economizar largura horizontal
    function formatarNaturezaDuasLinhas(val) {
      if (!val || String(val).trim() === '') return '';
      const s = String(val).trim();
      if (s.length <= 16) return s;
      const palavras = s.split(/\s+/);
      if (palavras.length <= 1) return s;
      const meio = Math.ceil(palavras.length / 2);
      return `${palavras.slice(0, meio).join(' ')}\n${palavras.slice(meio).join(' ')}`;
    }

    function getCategoriaVtrStat(nomeVtr) {
      const nome = String(nomeVtr || '').trim().toUpperCase();
      const match = viaturasStatusMap.find(v => String(v[0]).trim().toUpperCase() === nome);
      const cat = match ? String(match[1] || '').toUpperCase() : '';

      if (cat === 'RESGATE' || nome.startsWith('UR')) return 'RESGATE';
      if (['INCÊNDIO', 'INCENDIO', 'SALVAMENTO', 'INCÊNDIO/SALVAMENTO', 'SUPORTE AVANÇADO'].includes(cat) || nome.startsWith('ABT') || nome.startsWith('ABS')) {
        return 'INCENDIO';
      }
      return 'ADM';
    }

    function popularFiltrosEstatisticas() {
      const selectAno = document.getElementById("filtro-stat-ano");
      const selectVtr = document.getElementById("filtro-stat-viatura");
      if (!selectVtr) return;

      const anoAtual = selectAno ? selectAno.value : "";
      const vtrAtual = selectVtr ? selectVtr.value : "";

      // Anos disponíveis extraídos dos registros
      if (selectAno) {
        const anosSet = new Set();
        if (dadosTabela && dadosTabela.length > 1) {
          dadosTabela.slice(1).forEach(r => {
            if (!r) return;
            const dt = parseDateTimeFormat(r[8]);
            if (dt) anosSet.add(String(dt.getFullYear()));
          });
        }
        const listaAnos = Array.from(anosSet).sort().reverse();
        selectAno.innerHTML = `<option value="">Todos os anos</option>`;
        listaAnos.forEach(ano => {
          const opt = document.createElement("option");
          opt.value = ano;
          opt.textContent = ano;
          if (ano === anoAtual) opt.selected = true;
          selectAno.appendChild(opt);
        });
      }

      // Viaturas disponíveis
      const vtrsSet = new Set();
      if (viaturasLista) viaturasLista.forEach(v => { if (v) vtrsSet.add(v); });
      if (viaturasStatusMap) viaturasStatusMap.forEach(v => { if (v && v[0]) vtrsSet.add(v[0]); });
      if (dadosTabela && dadosTabela.length > 1) {
        dadosTabela.slice(1).forEach(r => { if (r && r[1]) vtrsSet.add(String(r[1]).trim()); });
      }

      const listaVtrs = Array.from(vtrsSet).sort();
      selectVtr.innerHTML = `<option value="">Todas as viaturas (${listaVtrs.length})</option>`;
      listaVtrs.forEach(v => {
        const opt = document.createElement("option");
        opt.value = v;
        opt.textContent = v;
        if (v === vtrAtual) opt.selected = true;
        selectVtr.appendChild(opt);
      });
    }

    function atualizarEstatisticasView() {
      try {
        // Preserva e sincroniza filtros diretamente a partir dos inputs na tela
        const elDtIni = document.getElementById("filtro-stat-data-inicio");
        const elDtFim = document.getElementById("filtro-stat-data-fim");
        const elAno = document.getElementById("filtro-stat-ano");
        const elMes = document.getElementById("filtro-stat-mes");
        const elDia = document.getElementById("filtro-stat-dia");
        const elVtr = document.getElementById("filtro-stat-viatura");
        const elCat = document.getElementById("filtro-stat-categoria");

        if (elDtIni && elDtIni.value !== undefined) filtroStatDataInicio = elDtIni.value;
        if (elDtFim && elDtFim.value !== undefined) filtroStatDataFim = elDtFim.value;
        if (elAno && elAno.value !== undefined) filtroStatAno = elAno.value;
        if (elMes && elMes.value !== undefined) filtroStatMes = elMes.value;
        if (elDia && elDia.value !== undefined) filtroStatDia = elDia.value;
        if (elVtr && elVtr.value !== undefined) filtroStatVtr = elVtr.value;
        if (elCat && elCat.value !== undefined) filtroStatCat = elCat.value;

        const temFiltroAtivo = Boolean(
          filtroStatDataInicio || filtroStatDataFim || filtroStatAno ||
          filtroStatMes || filtroStatDia || filtroStatVtr || filtroStatCat
        );

        const badgeFiltros = document.getElementById("badge-stat-filtros-ativos");
        if (badgeFiltros) {
          badgeFiltros.style.display = temFiltroAtivo ? "inline-block" : "none";
        }

        const subTitulo = document.getElementById("stat-chart-subtitle");
        if (subTitulo) {
          if (filtroStatVtr) {
            subTitulo.textContent = `Trajetória cronológica de quilometragem da viatura ${filtroStatVtr}`;
          } else {
            subTitulo.textContent = `Trajetória cronológica dos deslocamentos no período selecionado`;
          }
        }

        if (!dadosTabela || dadosTabela.length <= 1) {
          const elKpiDesloc = document.getElementById("stat-kpi-deslocamentos");
          const elKpiKm = document.getElementById("stat-kpi-km-total");
          const elKpiMedia = document.getElementById("stat-kpi-media-km");
          const elKpiOcorr = document.getElementById("stat-kpi-ocorrencias");
          const elKpiVtrs = document.getElementById("stat-kpi-viaturas-ativas");
          if (elKpiDesloc) elKpiDesloc.textContent = "0";
          if (elKpiKm) elKpiKm.textContent = "0 km";
          if (elKpiMedia) elKpiMedia.textContent = "0 km";
          if (elKpiOcorr) elKpiOcorr.textContent = "0";
          if (elKpiVtrs) elKpiVtrs.textContent = "0";
          renderizarGraficoEvolucaoKm([]);
          return;
        }

        const dtInicio = filtroStatDataInicio ? new Date(filtroStatDataInicio + "T00:00:00") : null;
        const dtFim = filtroStatDataFim ? new Date(filtroStatDataFim + "T23:59:59") : null;

        // Filtrar registros
        const registrosFiltradosStat = dadosTabela.slice(1).filter(r => {
          if (!r) return false;
          const vtr = String(r[1] || '').trim();
          const cat = getCategoriaVtrStat(vtr);

          if (filtroStatVtr && vtr.toUpperCase() !== filtroStatVtr.toUpperCase()) return false;
          if (filtroStatCat && cat !== filtroStatCat) return false;

          const dtSaida = parseDateTimeFormat(r[8]);
          if (!dtSaida) {
            if (dtInicio || dtFim || filtroStatAno || filtroStatMes || filtroStatDia) return false;
            return true;
          }

          if (dtInicio && dtSaida < dtInicio) return false;
          if (dtFim && dtSaida > dtFim) return false;

          if (filtroStatAno && dtSaida.getFullYear() !== Number(filtroStatAno)) return false;

          if (filtroStatMes) {
            const mesNum = String(dtSaida.getMonth() + 1).padStart(2, '0');
            if (mesNum !== filtroStatMes) return false;
          }

          if (filtroStatDia) {
            const ymd = `${dtSaida.getFullYear()}-${String(dtSaida.getMonth() + 1).padStart(2, '0')}-${String(dtSaida.getDate()).padStart(2, '0')}`;
            if (ymd !== filtroStatDia) return false;
          }

          return true;
        });

        // Cálculos Gerais
        let totalKm = 0;
        let totalDeslocamentos = registrosFiltradosStat.length;
        let viaturasUsadas = new Set();
        let ocorrenciasCount = 0;

        registrosFiltradosStat.forEach(r => {
          const vtr = String(r[1] || '');
          if (vtr) viaturasUsadas.add(vtr);

          const destino = String(r[4] || '').toUpperCase();
          if (destino === 'OCORRÊNCIA' || destino === 'OCORRENCIA') ocorrenciasCount++;

          const kmS = Number(r[9]);
          const kmC = Number(r[11]);
          if (!isNaN(kmS) && !isNaN(kmC) && kmC >= kmS) {
            totalKm += (kmC - kmS);
          }
        });

        const mediaKm = totalDeslocamentos > 0 ? (totalKm / totalDeslocamentos).toFixed(1) : "0";

        // Atualizar KPIs
        const elKpiDesloc = document.getElementById("stat-kpi-deslocamentos");
        const elKpiKm = document.getElementById("stat-kpi-km-total");
        const elKpiMedia = document.getElementById("stat-kpi-media-km");
        const elKpiOcorr = document.getElementById("stat-kpi-ocorrencias");
        const elKpiVtrs = document.getElementById("stat-kpi-viaturas-ativas");

        if (elKpiDesloc) elKpiDesloc.textContent = totalDeslocamentos;
        if (elKpiKm) elKpiKm.textContent = `${totalKm.toLocaleString('pt-BR')} km`;
        if (elKpiMedia) elKpiMedia.textContent = `${mediaKm} km`;
        if (elKpiOcorr) elKpiOcorr.textContent = ocorrenciasCount;
        if (elKpiVtrs) elKpiVtrs.textContent = viaturasUsadas.size;

        // 1. Agrupamento por TIPO (Destino)
        const mapaTipo = {};
        registrosFiltradosStat.forEach(r => {
          const rawTipo = String(r[4] || 'NÃO ESPECIFICADO').trim();
          const tipo = rawTipo || 'NÃO ESPECIFICADO';
          if (!mapaTipo[tipo]) mapaTipo[tipo] = { tipo, count: 0, km: 0 };
          mapaTipo[tipo].count++;

          const kmS = Number(r[9]);
          const kmC = Number(r[11]);
          if (!isNaN(kmS) && !isNaN(kmC) && kmC >= kmS) {
            mapaTipo[tipo].km += (kmC - kmS);
          }
        });

        const listaTipos = Object.values(mapaTipo).map(item => {
          const pct = totalDeslocamentos > 0 ? (item.count / totalDeslocamentos) * 100 : 0;
          const media = item.count > 0 ? (item.km / item.count) : 0;
          return { ...item, pct, media };
        });

        // Ordenar Lista de Tipos
        listaTipos.sort((a, b) => {
          let valA = a[sortStatTipo.col];
          let valB = b[sortStatTipo.col];
          if (typeof valA === 'string') {
            return sortStatTipo.asc ? valA.localeCompare(valB) : valB.localeCompare(valA);
          }
          return sortStatTipo.asc ? valA - valB : valB - valA;
        });

        // Atualizar ícones de ordenação tipo
        ['tipo', 'count', 'pct', 'km', 'media'].forEach(colName => {
          const iconEl = document.getElementById(`sort-icon-tipo-${colName}`);
          if (iconEl) {
            if (sortStatTipo.col === colName) {
              iconEl.textContent = sortStatTipo.asc ? '▲' : '▼';
              iconEl.style.color = 'var(--primary)';
            } else {
              iconEl.textContent = '↕';
              iconEl.style.color = 'var(--slate-400)';
            }
          }
        });

        const subtabLabelTipo = document.getElementById("subtab-label-tipo");
        if (subtabLabelTipo) subtabLabelTipo.textContent = `Por Tipo (${listaTipos.length})`;

        const tbodyTipo = document.getElementById("stat-tbody-tipo");
        if (tbodyTipo) {
          tbodyTipo.innerHTML = "";
          if (listaTipos.length === 0) {
            tbodyTipo.innerHTML = `<tr><td colspan="5" style="text-align: center; padding: 24px; color: var(--slate-400);">Nenhum deslocamento encontrado para os filtros.</td></tr>`;
          } else {
            listaTipos.forEach(item => {
              const isOcorr = item.tipo.toUpperCase() === 'OCORRÊNCIA' || item.tipo.toUpperCase() === 'OCORRENCIA';
              const bulletColor = isOcorr ? '#dc2626' : '#d97706';

              const tr = document.createElement("tr");
              tr.innerHTML = `
                <td style="font-weight: 600; color: var(--slate-800); display: flex; align-items: center; gap: 8px;">
                  <span style="width: 8px; height: 8px; border-radius: 50%; background: ${bulletColor}; flex-shrink: 0;"></span>
                  <span>${item.tipo}</span>
                </td>
                <td style="text-align: center; font-weight: 700; color: var(--slate-900);">${item.count}</td>
                <td style="text-align: center; font-weight: 600; color: var(--slate-600);">${item.pct.toFixed(1)}%</td>
                <td style="text-align: right; font-weight: 700; color: var(--emerald-600);">${item.km.toLocaleString('pt-BR')} km</td>
                <td style="text-align: right; color: var(--slate-600); font-weight: 500;">${item.media.toFixed(1)} km</td>
              `;
              tbodyTipo.appendChild(tr);
            });
          }
        }

        // 2. Agrupamento por CATEGORIA
        const mapaCat = {
          RESGATE: { id: 'RESGATE', label: 'Resgate (UR / APH)', count: 0, km: 0, viaturas: new Set(), icon: 'airport_shuttle' },
          INCENDIO: { id: 'INCENDIO', label: 'Incêndio & Salvamento (ABT / ABS)', count: 0, km: 0, viaturas: new Set(), icon: 'fire_truck' },
          ADM: { id: 'ADM', label: 'Apoio & Administrativas (ASA / ADM)', count: 0, km: 0, viaturas: new Set(), icon: 'minor_crash' }
        };

        registrosFiltradosStat.forEach(r => {
          const vtr = String(r[1] || '').trim();
          const cat = getCategoriaVtrStat(vtr);
          const targetKey = mapaCat[cat] ? cat : 'ADM';
          mapaCat[targetKey].count++;
          if (vtr) mapaCat[targetKey].viaturas.add(vtr);

          const kmS = Number(r[9]);
          const kmC = Number(r[11]);
          if (!isNaN(kmS) && !isNaN(kmC) && kmC >= kmS) {
            mapaCat[targetKey].km += (kmC - kmS);
          }
        });

        const listaCat = Object.values(mapaCat);
        listaCat.sort((a, b) => {
          let valA = a[sortStatCategoria.col];
          let valB = b[sortStatCategoria.col];
          if (typeof valA === 'string') {
            return sortStatCategoria.asc ? valA.localeCompare(valB) : valB.localeCompare(valA);
          }
          return sortStatCategoria.asc ? valA - valB : valB - valA;
        });

        ['count', 'km', 'label'].forEach(colName => {
          const iconEl = document.getElementById(`sort-icon-cat-${colName}`);
          if (iconEl) {
            if (sortStatCategoria.col === colName) {
              iconEl.textContent = sortStatCategoria.asc ? '▲' : '▼';
              iconEl.style.color = 'var(--primary)';
            } else {
              iconEl.textContent = '↕';
              iconEl.style.color = 'var(--slate-400)';
            }
          }
        });

        const containerCat = document.getElementById("stat-container-categoria");
        if (containerCat) {
          containerCat.innerHTML = "";
          listaCat.forEach(cat => {
            const pct = totalDeslocamentos > 0 ? ((cat.count / totalDeslocamentos) * 100).toFixed(1) : "0";
            const card = document.createElement("div");
            card.style.cssText = "background: var(--slate-50); border: 1px solid var(--slate-200); border-radius: var(--radius-md); padding: 16px; display: flex; flex-direction: column; justify-content: space-between;";
            card.innerHTML = `
              <div>
                <div style="display: flex; align-items: center; justify-content: space-between; margin-bottom: 10px;">
                  <div style="width: 38px; height: 38px; border-radius: 8px; background: white; border: 1px solid var(--slate-200); color: var(--primary); display: flex; align-items: center; justify-content: center;">
                    <span class="material-symbols-outlined">${cat.icon}</span>
                  </div>
                  <span style="font-size: 0.72rem; font-weight: 700; color: var(--slate-600); background: white; padding: 3px 8px; border-radius: 6px; border: 1px solid var(--slate-200);">
                    ${cat.viaturas.size} Vtrs
                  </span>
                </div>
                <h4 style="font-size: 0.9rem; font-weight: 800; color: var(--slate-900); margin-bottom: 2px;">${cat.label}</h4>
                <p style="font-size: 0.75rem; color: var(--slate-500); margin-bottom: 14px;">${pct}% de todas as saídas operacionais</p>
              </div>
              <div style="display: grid; grid-template-columns: 1fr 1fr; gap: 8px; text-align: center; padding-top: 10px; border-top: 1px solid var(--slate-200);">
                <div style="background: white; padding: 8px; border-radius: 6px; border: 1px solid var(--slate-200);">
                  <div style="font-size: 1.15rem; font-weight: 800; color: var(--slate-900); line-height: 1;">${cat.count}</div>
                  <div style="font-size: 0.65rem; font-weight: 700; color: var(--slate-500); text-transform: uppercase; margin-top: 2px;">Saídas</div>
                </div>
                <div style="background: white; padding: 8px; border-radius: 6px; border: 1px solid var(--slate-200);">
                  <div style="font-size: 1.15rem; font-weight: 800; color: var(--emerald-600); line-height: 1;">${cat.km.toLocaleString('pt-BR')} km</div>
                  <div style="font-size: 0.65rem; font-weight: 700; color: var(--slate-500); text-transform: uppercase; margin-top: 2px;">KM Total</div>
                </div>
              </div>
            `;
            containerCat.appendChild(card);
          });

        const mapaVtr = {};
        registrosFiltradosStat.forEach(r => {
          const vtr = String(r[1] || 'DESCONHECIDA').trim();
          if (!mapaVtr[vtr]) {
            mapaVtr[vtr] = {
              viatura: vtr,
              categoria: getCategoriaVtrStat(vtr),
              count: 0,
              kmRodado: 0,
              ultimoKm: 0
            };
          }
          mapaVtr[vtr].count++;

          const kmS = Number(r[9]);
          const kmC = Number(r[11]);
          if (!isNaN(kmS) && !isNaN(kmC) && kmC >= kmS) {
            mapaVtr[vtr].kmRodado += (kmC - kmS);
            if (kmC > mapaVtr[vtr].ultimoKm) mapaVtr[vtr].ultimoKm = kmC;
          } else if (!isNaN(kmS) && kmS > mapaVtr[vtr].ultimoKm) {
            mapaVtr[vtr].ultimoKm = kmS;
          }
        });

        const listaVtrs = Object.values(mapaVtr);

        // Ordenação de Viaturas
        listaVtrs.sort((a, b) => {
          let valA = a[sortStatViatura.col];
          let valB = b[sortStatViatura.col];
          if (typeof valA === 'string') {
            return sortStatViatura.asc ? valA.localeCompare(valB) : valB.localeCompare(valA);
          }
          return sortStatViatura.asc ? valA - valB : valB - valA;
        });

              iconEl.style.color = 'var(--slate-400)';

                  </button>
                </td>
              `;
              tbodyVtr.appendChild(tr);
            });
          }
        }

        // 4. Preparação e renderização do Gráfico de Linha
        const pontosGrafico = [];
        registrosFiltradosStat.forEach(r => {
          const dt = parseDateTimeFormat(r[8]);
          if (!dt) return;

          const vtr = String(r[1] || '');
          const destino = String(r[4] || '');
          const kmS = Number(r[9]);
          const kmC = Number(r[11]);

          let kmOdometro = (!isNaN(kmC) && kmC > 0) ? kmC : (!isNaN(kmS) ? kmS : 0);
          let kmRodado = (!isNaN(kmS) && !isNaN(kmC) && kmC >= kmS) ? (kmC - kmS) : 0;

          if (kmOdometro > 0 || kmRodado > 0) {
            pontosGrafico.push({
              id: r[0],
              dataHora: dt,
              dataHoraStr: `${String(dt.getDate()).padStart(2, '0')}/${String(dt.getMonth() + 1).padStart(2, '0')} ${String(dt.getHours()).padStart(2, '0')}:${String(dt.getMinutes()).padStart(2, '0')}`,
              viatura: vtr,
              destino: destino,
              kmOdometro: kmOdometro,
              kmRodado: kmRodado
            });
          }
        });

        pontosGrafico.sort((a, b) => a.dataHora.getTime() - b.dataHora.getTime());
        renderizarGraficoEvolucaoKm(pontosGrafico);
      } catch (err) {
        console.error("Erro ao atualizar visualização de estatísticas:", err);
      }
    }

    function renderizarGraficoEvolucaoKm(pontos) {
      const container = document.getElementById("stat-chart-container");
      const tooltip = document.getElementById("stat-chart-tooltip");
      if (!container) return;

      if (!pontos || pontos.length === 0) {
        container.innerHTML = `
          <div style="text-align: center; padding: 48px 16px; color: var(--slate-400);">
            <span class="material-symbols-outlined" style="font-size: 42px; color: var(--slate-300);">timeline</span>
            <p style="margin-top: 8px; font-size: 0.85rem; font-weight: 600;">Nenhum dado com quilometragem encontrado para os filtros selecionados.</p>
          </div>
        `;
        return;
      }

      const width = Math.max(800, container.clientWidth || 800);
      const height = 280;
      const padLeft = 70;
      const padRight = 30;
      const padTop = 25;
      const padBottom = 45;

      const chartW = width - padLeft - padRight;
      const chartH = height - padTop - padBottom;

      const valoresY = pontos.map(p => modoStatKm === 'registrado' ? p.kmOdometro : p.kmRodado);
      const minY = modoStatKm === 'rodado' ? 0 : Math.min(...valoresY);
      const maxY = Math.max(...valoresY);
      const rangeY = (maxY - minY) === 0 ? 1 : (maxY - minY);

      const coords = pontos.map((p, idx) => {
        const x = pontos.length === 1 ? (padLeft + chartW / 2) : (padLeft + (idx / (pontos.length - 1)) * chartW);
        const val = modoStatKm === 'registrado' ? p.kmOdometro : p.kmRodado;
        const y = padTop + chartH - ((val - minY) / rangeY) * chartH;
        return { x, y, raw: p, valor: val };
      });

      const pathLine = coords.map((pt, i) => `${i === 0 ? 'M' : 'L'} ${pt.x.toFixed(1)} ${pt.y.toFixed(1)}`).join(" ");
      const pathArea = `${pathLine} L ${coords[coords.length - 1].x.toFixed(1)} ${padTop + chartH} L ${coords[0].x.toFixed(1)} ${padTop + chartH} Z`;

      let svgHtml = `
        <svg viewBox="0 0 ${width} ${height}" style="width: 100%; height: auto; min-width: 700px; overflow: visible; font-family: inherit;">
          <defs>
            <linearGradient id="gasGradArea" x1="0" y1="0" x2="0" y2="1">
              <stop offset="0%" stop-color="#991b1b" stop-opacity="0.25" />
              <stop offset="100%" stop-color="#991b1b" stop-opacity="0.0" />
            </linearGradient>
          </defs>
      `;

      // Linhas de Grade Horizontais e Labels Y
      [0, 0.25, 0.5, 0.75, 1].forEach((pct, i) => {
        const y = padTop + chartH - (pct * chartH);
        const val = Math.round(minY + pct * rangeY);
        const dash = (i === 0 || i === 4) ? '' : 'stroke-dasharray="3,3"';
        svgHtml += `
          <line x1="${padLeft}" y1="${y}" x2="${width - padRight}" y2="${y}" stroke="#e2e8f0" ${dash} />
          <text x="${padLeft - 10}" y="${y + 4}" text-anchor="end" font-size="10" font-weight="600" fill="#64748b">${val.toLocaleString('pt-BR')} km</text>
        `;
      });

      // Área preenchida
      svgHtml += `<path d="${pathArea}" fill="url(#gasGradArea)" />`;

      // Linha principal
      svgHtml += `<path d="${pathLine}" fill="none" stroke="#991b1b" stroke-width="2.5" stroke-linecap="round" stroke-linejoin="round" />`;

      // Pontos e Labels X
      coords.forEach((pt, idx) => {
        svgHtml += `
          <circle cx="${pt.x.toFixed(1)}" cy="${pt.y.toFixed(1)}" r="4.5" fill="#ffffff" stroke="#991b1b" stroke-width="2" style="cursor: pointer;"
            data-idx="${idx}">
          </circle>
        `;

        if (idx === 0 || idx === coords.length - 1 || idx === Math.floor(coords.length / 2)) {
          svgHtml += `
            <text x="${pt.x.toFixed(1)}" y="${padTop + chartH + 20}" text-anchor="middle" font-size="10" font-weight="600" fill="#64748b">
              ${pt.raw.dataHoraStr}
            </text>
          `;
        }
      });

      svgHtml += `</svg>`;
      container.innerHTML = svgHtml;

      // Eventos interativos de hover nos pontos
      const circles = container.querySelectorAll("circle");
      circles.forEach(c => {
        c.addEventListener("mouseenter", function(e) {
          const idx = parseInt(this.getAttribute("data-idx"), 10);
          const pt = coords[idx];
          if (!pt || !tooltip) return;

          this.setAttribute("r", "6.5");
          this.setAttribute("stroke-width", "3");

          const rotuloValor = modoStatKm === 'registrado' ? 'KM Odômetro' : 'KM Percorrido';
          tooltip.innerHTML = `
            <div style="font-weight: 800; color: #fca5a5; margin-bottom: 2px;">${pt.raw.viatura}</div>
            <div style="font-size: 0.7rem; color: #cbd5e1;">${pt.raw.dataHoraStr}</div>
            <div style="font-size: 0.7rem; color: #cbd5e1; max-width: 180px; overflow: hidden; text-overflow: ellipsis; white-space: nowrap;">${pt.raw.destino}</div>
            <div style="font-weight: 800; color: white; margin-top: 4px; padding-top: 4px; border-top: 1px solid #334155;">
              ${rotuloValor}: ${pt.valor.toLocaleString('pt-BR')} km
            </div>
          `;
          tooltip.style.display = "block";

          const containerRect = container.getBoundingClientRect();
          const circleRect = this.getBoundingClientRect();
          const leftPos = circleRect.left - containerRect.left + (circleRect.width / 2);
          const topPos = circleRect.top - containerRect.top;

          tooltip.style.left = `${leftPos}px`;
          tooltip.style.top = `${topPos}px`;
        });

        c.addEventListener("mouseleave", function() {
          this.setAttribute("r", "4.5");
          this.setAttribute("stroke-width", "2");
          if (tooltip) tooltip.style.display = "none";
        });
      });
    }
